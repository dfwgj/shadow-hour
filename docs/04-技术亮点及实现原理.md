## 一、跨平台 HTTP 流式处理

### 1.1 技术亮点

在移动端实现 LLM 流式响应需要解决 **NativeScript 没有原生 SSE 支持** 的问题。本项目实现了完全原生的跨平台流式 HTTP 客户端。

### 1.2 实现原理

#### 平台自动选择机制

```typescript
// 文件命名约定实现平台选择
```

#### Android 实现 (OkHttp Worker)

```typescript
// stream.android.ts
// 使用 Worker 线程避免阻塞 UI

@JavaProxy("org.example.HttpWorker")
class HttpWorker extends java.lang.Thread {
  run() {
    const client = new okhttp3.OkHttpClient.Builder()
      .readTimeout(0, java.util.concurrent.TimeUnit.SECONDS)  // 无限超时
      .build();

    const request = new okhttp3.Request.Builder()
      .url(url)
      .post(requestBody)
      .build();

    const response = client.newCall(request).execute();
    const source = response.body().source();

    // 逐行读取 SSE 数据
    while (!source.exhausted()) {
      const line = source.readUtf8Line();
      // 通过消息传递回主线程
      postMessage({ type: 'data', line });
    }
  }
}
```

#### iOS 实现 (NSURLSession)

```typescript
// stream.ios.ts
// 使用 NSURLSessionDataDelegate 处理流式数据

class StreamDelegate extends NSObject implements NSURLSessionDataDelegate {
  URLSessionDataTaskDidReceiveData(session, task, data) {
    const chunk = NSString.alloc()
      .initWithDataEncoding(data, NSUTF8StringEncoding);
    // 回调处理数据
    this.onData(chunk);
  }

  URLSessionTaskDidCompleteWithError(session, task, error) {
    if (error) {
      this.onError(error);
    } else {
      this.onComplete();
    }
  }
}
```

### 1.3 核心优势

- **真正的原生性能**：不依赖 WebView 或 JS polyfill
- **内存效率高**：逐行处理，不需要缓存整个响应
- **可中断**：支持随时中止请求
- **超时控制**：可配置请求超时

### 3.2 实现原理

#### Skill 定义格式

#### 渐进式加载

```typescript
enum SkillLoadLevel {
  METADATA = 1,  // 仅加载 frontmatter (用于搜索)
  CONTENT = 2,   // 加载主体内容
  FULL = 3       // 加载完整内容 (含参考资料)
}

class SkillRegistry {
  private skills = new Map<string, Skill>();

  // 按需加载
  loadSkill(id: string, level: SkillLoadLevel) {
    const skill = this.skills.get(id);

    if (skill.loadLevel >= level) return skill;

    // 根据 level 加载更多内容
    if (level >= SkillLoadLevel.CONTENT) {
      skill.content = this.parseContent(skill.raw);
    }

    skill.loadLevel = level;
    return skill;
  }
}
```

#### 智能匹配算法

```typescript
function getSkillsPrompt(userInput: string, maxSkills: number): string {
  const scores = [];

  for (const skill of registry.getAllSkills()) {
    // 基于标签匹配
    const tagScore = calculateTagMatch(userInput, skill.tags);

    // 基于描述匹配
    const descScore = calculateTextSimilarity(userInput, skill.description);

    // 优先级加权
    const finalScore = (tagScore + descScore) * skill.priority;

    scores.push({ skill, score: finalScore });
  }

  // 取 Top N
  return scores
    .sort((a, b) => b.score - a.score)
    .slice(0, maxSkills)
    .map(s => s.skill.content)
    .join('\n\n');
}
```

### 3.3 核心优势

- **模块化**：技能可独立开发和维护
- **低成本**：渐进式加载减少 Token 消耗
- **可扩展**：支持自定义技能包

#### 工具注册与执行

```typescript
class ToolRegistry {
  private tools = new Map<string, {
    definition: ToolDefinition;
    handler: ToolHandler;
  }>();

  register(definition: ToolDefinition, handler: ToolHandler) {
    this.tools.set(definition.name, { definition, handler });
  }

  async execute(
    name: string,
    input: unknown,
    context: ToolContext
  ): Promise<ToolResult> {
    const tool = this.tools.get(name);

    if (!tool) {
      return { success: false, content: '', error: `Unknown tool: ${name}` };
    }

    try {
      const result = await tool.handler(input, context);
      return { success: true, content: JSON.stringify(result) };
    } catch (error) {
      return { success: false, content: '', error: error.message };
    }
  }
}
```

#### 流式工具调用处理

```typescript
// 解析流式响应中的工具调用
function handleStreamEvent(event: StreamEvent) {
  if (event.type === 'tool_use_start') {
    // 开始累积工具调用
    currentToolCall = {
      id: event.data.id,
      name: event.data.name,
      arguments: ''
    };
  }

  if (event.type === 'tool_use_delta') {
    // 累积 JSON 参数
    currentToolCall.arguments += event.data.partial_json;
  }

  if (event.type === 'tool_use_stop') {
    // 解析并执行
    const args = JSON.parse(currentToolCall.arguments);
    const result = await toolRegistry.execute(
      currentToolCall.name,
      args,
      context
    );

    // 将结果返回给 LLM
    return result;
  }
}
```

- **可扩展**：轻松添加新工具

## 五、四层事件驱动架构

### 5.1 技术亮点

将 AI 交互分为四层事件系统，实现细粒度的状态管理和事件处理。

### 5.2 实现原理

#### 事件层次定义

```typescript
// Layer 1: 流事件 (最底层)
type StreamEvent =
  | { type: 'message_start'; data: { messageId: string } }
  | { type: 'text_delta'; data: { text: string } }
  | { type: 'tool_use_start'; data: { toolCallId: string; toolName: string } }
  | { type: 'tool_use_delta'; data: { partialJson: string } }
  | { type: 'tool_use_stop'; data: { toolCallId: string } }
  | { type: 'message_stop'; data: { stopReason: string } }
  | { type: 'stream_error'; data: { error: string } };

// Layer 2: 状态事件
type StateEvent =
  | { type: 'state_change'; data: { from: AgentState; to: AgentState } }
  | { type: 'conversation_start' }
  | { type: 'conversation_end' };

// Layer 3: 工具事件
type ToolEvent =
  | { type: 'tool_call_start'; data: ToolCall }
  | { type: 'tool_call_result'; data: ToolResult }
  | { type: 'tool_call_error'; data: { error: string } };

// Layer 4: Agent 状态
type AgentState =
  | 'idle'        // 空闲
  | 'thinking'    // 思考中
  | 'responding'  // 响应中
  | 'tool_calling'// 调用工具
  | 'interrupted' // 已中断
  | 'error';      // 错误
```

#### 状态机实现 (Mealy Machine)

```typescript
// 状态转换函数
function processStreamEvent(
  state: AgentState,
  event: StreamEvent
): [AgentState, OutputEvent[]] {
  const outputs: OutputEvent[] = [];

  switch (event.type) {
    case 'message_start':
      return ['responding', [{ type: 'state_change', ... }]];

    case 'text_delta':
      return [state, [event]];  // 透传文本事件

    case 'tool_use_start':
      return ['tool_calling', [{ type: 'state_change', ... }, event]];

    case 'message_stop':
      return ['idle', [{ type: 'state_change', ... }]];

    case 'stream_error':
      return ['error', [{ type: 'state_change', ... }]];
  }
}
```

#### 事件总线

```typescript
class EventBus {
  private handlers = new Set<EventHandler>();

  on(handler: EventHandler): () => void {
    this.handlers.add(handler);
    return () => this.handlers.delete(handler);
  }

  emit(event: AgentEvent) {
    for (const handler of this.handlers) {
      handler(event);
    }
  }
}
```

### 5.3 核心优势

- **解耦**：各层独立处理自己的事件
- **可追溯**：完整的事件流便于调试
- **可扩展**：轻松添加新的事件类型

## 六、RFC5545 iCalendar 完整实现

### 6.1 技术亮点

完整实现 iCalendar 标准（RFC5545），支持重复规则、提醒、多时区等特性。

### 6.2 实现原理

#### 类型定义

```typescript
// VEVENT 事件
interface CalendarEvent {
  uid: string;           // 唯一标识
  dtStart: Date;         // 开始时间 (DTSTART)
  dtEnd?: Date;          // 结束时间 (DTEND)
  summary: string;       // 标题 (SUMMARY)
  description?: string;  // 描述 (DESCRIPTION)
  location?: string;     // 地点 (LOCATION)
  status?: EventStatus;  // 状态 (STATUS)
  rrule?: RecurrenceRule;// 重复规则 (RRULE)
  alarms?: VAlarm[];     // 提醒 (VALARM)
}

// RRULE 重复规则
interface RecurrenceRule {
  freq: 'DAILY' | 'WEEKLY' | 'MONTHLY' | 'YEARLY';
  interval?: number;     // 间隔
  count?: number;        // 次数
  until?: Date;          // 截止日期
  byday?: string[];      // 星期几 (MO,TU,WE...)
  bymonthday?: number[]; // 每月第几天
  bymonth?: number[];    // 第几月
}

// VALARM 提醒
interface VAlarm {
  action: 'DISPLAY' | 'AUDIO' | 'EMAIL';
  trigger: number;       // 提前时间 (分钟)
  description?: string;
}
```

#### 重复事件展开

```typescript
function expandRecurringEvent(
  event: CalendarEvent,
  start: Date,
  end: Date
): CalendarEvent[] {
  if (!event.rrule) return [event];

  const occurrences: CalendarEvent[] = [];
  let current = new Date(event.dtStart);
  let count = 0;

  while (current <= end) {
    if (current >= start) {
      occurrences.push({
        ...event,
        uid: `${event.uid}_${count}`,
        dtStart: new Date(current),
        dtEnd: event.dtEnd
          ? new Date(current.getTime() + (event.dtEnd.getTime() - event.dtStart.getTime()))
          : undefined
      });
    }

    // 根据 freq 计算下一个发生时间
    current = getNextOccurrence(current, event.rrule);
    count++;

    if (event.rrule.count && count >= event.rrule.count) break;
    if (event.rrule.until && current > event.rrule.until) break;
  }

  return occurrences;
}
```

### 6.3 核心优势

- **标准兼容**：可与其他日历应用互操作
- **功能完整**：支持复杂的重复规则
- **性能优化**：按需展开重复事件

```typescript
// useTheme.ts
export function useTheme() {
  const theme = ref<'light' | 'dark' | 'system'>('system');
  const isMidnight = ref(false);

  function updateTheme() {
    const hour = new Date().getHours();
    isMidnight.value = hour >= 23 || hour < 5;

    const isDark = theme.value === 'dark' ||
      (theme.value === 'system' && systemDarkMode());

    // 应用主题类
    const root = Application.getRootView();
    root.className = isDark
      ? (isMidnight.value ? 'theme-midnight' : 'theme-dark')
      : 'theme-light';
  }

  // 每分钟检查时间
  setInterval(updateTheme, 60000);

  return { theme, isMidnight, updateTheme };
}
```
